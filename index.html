// 🔗 투네이션 알림 토큰 설정 const MY_ALERT_TOKEN = "75c6b8a21c7823b99a405a00054b2e51"; // 🔄 [리셋 설정] 방송을 새로 켤 때마다 항상 0원, 0스택으로 깨끗하게 시작하도록 변경 let currentMoney = 0; let smokeStack = 0; let quitSmoke = 0; let eatCount = 0; let noEatCount = 0; // 숫자가 바뀔 때마다 질문자님의 원래 디자인 틀 엘리먼트에 값을 실시간 주입하는 함수 function updateUI() { if (document.getElementById('moneyText')) document.getElementById('moneyText').innerText = currentMoney.toLocaleString() + '원'; if (document.getElementById('smokeStack')) document.getElementById('smokeStack').innerText = smokeStack; if (document.getElementById('quitSmoke')) document.getElementById('quitSmoke').innerText = quitSmoke; if (document.getElementById('eatCount')) document.getElementById('eatCount').innerText = eatCount; if (document.getElementById('noEatCount')) document.getElementById('noEatCount').innerText = noEatCount; } // 위젯이 시작되자마자 즉시 모든 칸을 0으로 강제 세팅 updateUI(); // 🔗 질문자님이 원래 쓰시던 방식 그대로 소켓 서버 연결 (인증 규칙만 보완) const socket = io("https://toon.at", { path: "/socket.io", query: { token: MY_ALERT_TOKEN }, transports: ["websocket"] }); // 💰 [핵심 계산기] 금액 정산 및 1,000원 이상 후원의 단어 포함 여부 판별 구간 function processDonation(amount, comment = "") { if (!isNaN(amount) && amount > 0) { // 누적 용돈 추가 currentMoney += amount; // ⚡ 후원 금액이 1,000원 이상일 때만 후원 메시지 속 단어를 분석합니다. if (amount >= 1000) { if (comment.includes("흡연")) { smokeStack += 1; } if (comment.includes("금연")) { quitSmoke += 1; } if (comment.includes("먹어")) { eatCount += 1; } if (comment.includes("먹지마")) { noEatCount += 1; } } // 갱신된 점수를 질문자님의 원래 틀 화면에 반영 updateUI(); } } // ⚡ 실제 시청자 후원 실시간 수신 이벤트 처리 socket.on("donation", (data) => { if (data && data.amount) { processDonation(parseInt(data.amount), data.comment || ""); } }); // ⚡ 투네이션 크리에이터 페이지의 [테스트 후원] 버튼 신호 처리 socket.on("test", (data) => { if (data && data.amount) { // 테스트 메시지는 text나 comment로 들어올 수 있으므로 안전하게 병합 처리합니다. const testComment = data.comment || data.text || ""; processDonation(parseInt(data.amount), testComment); } });
